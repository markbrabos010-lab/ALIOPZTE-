async function approveReview(revId) { const token = localStorage.getItem("github_token") || document.getElementById("ghToken").value.trim(); if (!token) { alert("ضع مفتاح Token أولاً!"); return; } try { // جلب أحدث بيانات كاملة مباشرة من GitHub قبل الحفظ const getUrl = `https://api.github.com/repos/${GITHUB_USERNAME}/${REPO_NAME}/contents/${FILE_PATH}`; const res = await fetch(getUrl, { headers: { 'Authorization': `token ${token}` } }); if (!res.ok) throw new Error("تعذر جلب البيانات من GitHub"); const fileInfo = await res.json(); const binaryString = atob(fileInfo.content.replace(/\s/g, '')); const bytes = new Uint8Array(binaryString.length); for (let i = 0; i < binaryString.length; i++) bytes[i] = binaryString.charCodeAt(i); const decodedText = new TextDecoder('utf-8').decode(bytes); const onlineData = JSON.parse(decodedText); let itemsList = Array.isArray(onlineData) ? onlineData : (onlineData.items || []); let reviewsList = onlineData.reviews || []; let customBg = onlineData.customBg || ""; // تعديل حالة التقييم المختار فقط const rev = reviewsList.find(r => r.id === revId); if (rev) { rev.status = 'approved'; } // حفظ القائمة الكاملة دون مسح الألعاب أو الفيديوهات await updateGithubFile(token, itemsList, reviewsList, "قبول تقييم ونشره للعلن"); } catch (err) { alert("خطأ: " + err.message); } } async function replyReview(revId) { const token = localStorage.getItem("github_token") || document.getElementById("ghToken").value.trim(); if (!token) { alert("ضع مفتاح Token أولاً!"); return; } const replyText = prompt("اكتب ردك المباشر على التقييم/الطلب:"); if (!replyText) return; try { const getUrl = `https://api.github.com/repos/${GITHUB_USERNAME}/${REPO_NAME}/contents/${FILE_PATH}`; const res = await fetch(getUrl, { headers: { 'Authorization': `token ${token}` } }); if (!res.ok) throw new Error("تعذر جلب البيانات من GitHub"); const fileInfo = await res.json(); const binaryString = atob(fileInfo.content.replace(/\s/g, '')); const bytes = new Uint8Array(binaryString.length); for (let i = 0; i < binaryString.length; i++) bytes[i] = binaryString.charCodeAt(i); const decodedText = new TextDecoder('utf-8').decode(bytes); const onlineData = JSON.parse(decodedText); let itemsList = Array.isArray(onlineData) ? onlineData : (onlineData.items || []); let reviewsList = onlineData.reviews || []; const rev = reviewsList.find(r => r.id === revId); if (rev) { rev.reply = replyText; rev.status = 'approved'; } await updateGithubFile(token, itemsList, reviewsList, "الرد على تقييم ونشره للعلن"); } catch (err) { alert("خطأ: " + err.message); } }
